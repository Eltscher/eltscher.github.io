// Formular-Validierung
<?php
function validateForm($data) {
    $errors = [];

    // Überprüfen, ob Name ausgefüllt wurde
    if (empty($data['name'])) {
        $errors['name'] = "Name ist erforderlich.";
    }

    // Überprüfen, ob die E-Mail-Adresse gültig ist
    if (empty($data['email']) || !filter_var($data['email'], FILTER_VALIDATE_EMAIL)) {
        $errors['email'] = "Gültige E-Mail-Adresse ist erforderlich.";
    }

    // Überprüfen, ob Nachricht ausgefüllt wurde
    if (empty($data['message'])) {
        $errors['message'] = "Nachricht ist erforderlich.";
    }

    return $errors;
}

// Beispiel: Formulardaten validieren
$data = [
    'name' => $_POST['name'] ?? '',
    'email' => $_POST['email'] ?? '',
    'message' => $_POST['message'] ?? ''
];

$errors = validateForm($data);
if (empty($errors)) {
    // Daten sind gültig, weiterverarbeiten
} else {
    // Fehler ausgeben
    foreach ($errors as $field => $error) {
        echo "<p>$field: $error</p>";
    }
}
?>

// Daten in MySQL speichern
<?php
$servername = "localhost";
$username = "root";
$password = "";
$dbname = "testdb";

// Verbindung aufbauen
$conn = new mysqli($servername, $username, $password, $dbname);

// Verbindung prüfen
if ($conn->connect_error) {
    die("Verbindung fehlgeschlagen: " . $conn->connect_error);
}

// SQL-Statement zum Einfügen von Daten
$stmt = $conn->prepare("INSERT INTO users (name, email, message) VALUES (?, ?, ?)");
$stmt->bind_param("sss", $name, $email, $message);

// Formulardaten
$name = $_POST['name'];
$email = $_POST['email'];
$message = $_POST['message'];

// Ausführen
if ($stmt->execute()) {
    echo "Neuer Datensatz erfolgreich erstellt";
} else {
    echo "Fehler: " . $stmt->error;
}

// Verbindung schließen
$stmt->close();
$conn->close();
?>

// Session-Management
<?php
session_start();

// Benutzer anmelden
if (isset($_POST['username']) && isset($_POST['password'])) {
    $_SESSION['username'] = $_POST['username'];
    echo "Willkommen, " . $_SESSION['username'] . "!";
}

// Benutzer abmelden
if (isset($_GET['logout'])) {
    session_destroy();
    header("Location: index.php");
}

// Eingabeformular
if (!isset($_SESSION['username'])) {
    echo '<form method="POST">
            Benutzername: <input type="text" name="username" required><br>
            Passwort: <input type="password" name="password" required><br>
            <input type="submit" value="Anmelden">
          </form>';
} else {
    echo 'Eingeloggt als: ' . $_SESSION['username'] . ' <a href="?logout=true">Abmelden</a>';
}
?>
